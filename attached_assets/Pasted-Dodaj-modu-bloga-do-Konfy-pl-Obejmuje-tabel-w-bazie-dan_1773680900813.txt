━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

W pliku `server/routes.ts` dodaj endpointy bloga.
Publiczne (bez autoryzacji):

  GET /api/blog
  - zwraca listę postów ze status="published"
  - obsługuje query param ?specialization= do filtrowania
  - zwraca pola: id, title, slug, excerpt, metaTitle, metaDescription,
    coverImage, specialization, tags, authorName, readingTimeMinutes, publishedAt

  GET /api/blog/:slug
  - zwraca pojedynczy post po slug ze status="published"
  - 404 jeśli nie znaleziony lub draft

Chronione (isAdmin):

  GET /api/admin/blog
  - zwraca wszystkie posty (draft + published), sortowane createdAt DESC

  POST /api/admin/blog
  - tworzy nowy post
  - body: { title, slug, excerpt, content, metaTitle, metaDescription,
    coverImage?, specialization?, tags?, authorName?, readingTimeMinutes?,
    status, publishedAt? }
  - walidacja: title i content wymagane
  - auto-generuj slug z title jeśli slug pusty:
    slug = title.toLowerCase()
      .replace(/ą/g,'a').replace(/ę/g,'e').replace(/ó/g,'o')
      .replace(/ś/g,'s').replace(/ł/g,'l').replace(/ż/g,'z')
      .replace(/ź/g,'z').replace(/ć/g,'c').replace(/ń/g,'n')
      .replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '')
  - jeśli status="published" i brak publishedAt, ustaw publishedAt=new Date()

  PUT /api/admin/blog/:id
  - aktualizuje post
  - te same pola co POST
  - jeśli zmiana status z "draft" na "published" i brak publishedAt → ustaw now

  DELETE /api/admin/blog/:id
  - usuwa post, zwraca { success: true }

CZĘŚĆ C — funkcje fetch/save dla bloga:

const fetchBlogPosts = async () => {
  setBlogLoading(true);
  try {
    const res = await fetch("/api/admin/blog", { credentials: "include" });
    const data = await res.json();
    setBlogPosts(data);
  } finally {
    setBlogLoading(false);
  }
};

const saveBlogPost = async () => {
  const payload = {
    ...blogForm,
    tags: blogForm.tags.split(",").map(t => t.trim()).filter(Boolean),
    readingTimeMinutes: Number(blogForm.readingTimeMinutes),
  };
  const url = editingPost ? `/api/admin/blog/${editingPost.id}` : "/api/admin/blog";
  const method = editingPost ? "PUT" : "POST";
  const res = await fetch(url, {
    method, credentials: "include",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payload),
  });
  if (res.ok) {
    setBlogDialogOpen(false);
    setEditingPost(null);
    fetchBlogPosts();
    toast({ title: editingPost ? "Post zaktualizowany" : "Post utworzony" });
  }
};

const deleteBlogPost = async (id: number) => {
  await fetch(`/api/admin/blog/${id}`, { method: "DELETE", credentials: "include" });
  fetchBlogPosts();
};

const openNewPost = () => {
  setEditingPost(null);
  setBlogForm({
    title: "", slug: "", excerpt: "", content: "",
    metaTitle: "", metaDescription: "", coverImage: "",
    specialization: "", authorName: "Redakcja Konfy.pl",
    readingTimeMinutes: 5, status: "draft", tags: ""
  });
  setBlogDialogOpen(true);
};

const openEditPost = (post: any) => {
  setEditingPost(post);
  setBlogForm({
    title: post.title, slug: post.slug, excerpt: post.excerpt,
    content: post.content, metaTitle: post.metaTitle || "",
    metaDescription: post.metaDescription || "",
    coverImage: post.coverImage || "",
    specialization: post.specialization || "",
    authorName: post.authorName || "Redakcja Konfy.pl",
    readingTimeMinutes: post.readingTimeMinutes || 5,
    status: post.status, tags: (post.tags || []).join(", ")
  });
  setBlogDialogOpen(true);
};
